// ------------------------------------------------------------ détail function closeDetails() { state.selected = ''; clear($('details')); $('details').hidden = true; $('main').classList.remove('with-details'); Array.prototype.forEach.call(document.querySelectorAll('.selected'), function (n) { n.classList.remove('selected'); }); } function openDetails(nodes) { var a = $('details'); clear(a); a.appendChild(el('button', { type: 'button', class: 'close', 'aria-label': 'Fermer le détail', text: '×', onclick: closeDetails })); nodes.forEach(function (n) { if (n) a.appendChild(n); }); a.hidden = false; $('main').classList.add('with-details'); } function facts(pairs) { var dl = el('dl'); pairs.forEach(function (p) { if (p[1] === null || p[1] === undefined || p[1] === '') return; dl.appendChild(el('dt', { text: p[0] })); dl.appendChild(el('dd', { text: String(p[1]) })); }); return dl; } function markSelected(id) { state.selected = id; Array.prototype.forEach.call(document.querySelectorAll('.selected'), function (n) { n.classList.remove('selected'); }); var g = $('canvas').querySelector('[data-id="' + cssEscape(id) + '"]'); if (g) g.classList.add('selected'); } function cssEscape(s) { return window.CSS && CSS.escape ? CSS.escape(s) : String(s).replace(/["\\]/g, '\\$&'); } function showNode(id) { var n = state.data.nodes.filter(function (x) { return x.id === id; })[0]; if (!n) return; markSelected(id); var nodes = []; var kind = n.kind === 'program' ? 'Programme' : n.kind === 'project' ? 'Projet' : n.kind === 'wp' ? (n.parent_ref ? 'Sous-workpackage' : 'Workpackage') : n.item_type; nodes.push(el('h2', { text: (n.code ? n.code + ' ' : '') + n.name }), el('p', { class: 'kind', text: kind })); var flt = n.float === null || n.float === undefined ? '' : (n.float < 0 ? 'Décale un successeur de ' + plural(-n.float, 'jour ouvré', 'jours ouvrés') : n.float === 0 ? 'Aucune : chemin critique' : plural(n.float, 'jour ouvré', 'jours ouvrés')); nodes.push(facts([ ['Responsable', n.owner], ['Organisation', n.owner ? (n.owner_org || 'Non renseignée') : ''], ['Fonction', n.owner_function], ['Début', frDate(n.start)], ['Fin', frDate(n.finish)], ['Avancement', n.kind === 'item' && n.item_type === 'Jalon' ? '' : n.progress + ' %'], ['Statut', n.kind === 'item' && n.item_type === 'Jalon' ? (n.passed_on ? 'Passé le ' + frDate(n.passed_on) : (n.late ? 'Échéance dépassée : pas encore marqué passé' : 'À venir')) : n.status], ['Imputation', n.charge_code], ['Marge', flt] ])); if (n.late) nodes.push(el('p', { class: 'flag alert', text: 'Au moins un élément est en retard sur sa date de fin prévue.' })); if (state.data.scope === 'program') { if (n.kind !== 'program') nodes.push(openProjectActions(n)); } else { if (n.kind === 'project' || n.kind === 'wp' || n.kind === 'item') nodes.push(rolesBlock(n)); if (n.canEdit || n.canPass) nodes.push(wbsActions(n)); } if (n.kind === 'item' && state.data.scope !== 'program') nodes.push(depsBlock(n)); if (BOOT.appsheetUrl) nodes.push(el('div', { class: 'links' }, [el('a', { class: 'btn', href: BOOT.appsheetUrl, target: '_blank', rel: 'noopener', text: 'Modifier dans AppSheet' })])); openDetails(nodes); }
