// ------------------------------------------------------------ création et édition du WBS var KIND_NAMES = { wp: 'workpackage', Livrable: 'livrable', Jalon: 'jalon' }; function ensurePeople() { if (state.people) return Promise.resolve(state.people); return call('resources.list', { limit: 500 }).then(function (rows) { state.people = rows; return rows; }).catch(function () { return []; }); } function below(id) { var out = []; state.data.nodes.forEach(function (n) { if (n.parent === id) { out.push(n); out = out.concat(below(n.id)); } }); return out; } function selectPending() { var id = state.pendingSelect; state.pendingSelect = ''; if (id && state.data && state.data.nodes.some(function (n) { return n.id === id; })) showNode(id); } function closeDialog(d) { if (d.close) d.close(); else d.removeAttribute('open'); if (/^genform/.test(d.id)) clear($('gf-body')); if (/^editor/.test(d.id)) clear($('editor-body')); } /** Boutons du panneau de détail : seulement ceux que le serveur autorise pour cette carte. */ function wbsActions(n) { var box = el('div', { class: 'links', id: 'wbs-actions' }); var add = function (label, kind, id) { box.appendChild(el('button', { type: 'button', class: 'btn', id: id, text: label, onclick: function () { openEditor({ mode: 'create', kind: kind, parent: n }); } })); }; if (n.kind === 'project') { add('Ajouter un workpackage', 'wp', 'add-wp'); add('Ajouter un livrable', 'Livrable', 'add-deliverable'); add('Ajouter un jalon', 'Jalon', 'add-milestone'); } if (n.kind === 'wp') { if (!n.parent_ref) add('Ajouter un sous-workpackage', 'wp', 'add-subwp'); add('Ajouter un livrable', 'Livrable', 'add-deliverable'); add('Ajouter un jalon', 'Jalon', 'add-milestone'); } if (n.kind !== 'project') { box.appendChild(el('button', { type: 'button', class: 'btn primary', id: 'edit-node', text: 'Modifier', onclick: function () { openEditor({ mode: 'edit', kind: n.kind === 'wp' ? 'wp' : n.item_type, node: n }); } })); box.appendChild(el('button', { type: 'button', class: 'btn', id: 'delete-node', text: 'Supprimer', onclick: function () { openDelete(n); } })); } return box; } function parentChoices(kind, n) { var hasKids = kind === 'wp' && state.data.nodes.some(function (x) { return x.parent_ref === n.ref; }); var out = [{ value: '', label: 'Projet (racine)' }]; if (hasKids) return out; state.data.nodes.forEach(function (x) { if (x.kind !== 'wp') return; if (kind === 'wp' && (x.parent_ref || x.ref === n.ref)) return; out.push({ value: x.ref, label: (x.code ? x.code + ' ' : '') + x.name }); }); return out; } function openEditor(spec) { ensurePeople().then(function (people) { var create = spec.mode === 'create', kind = spec.kind, n = spec.node || {}; var isSub = create && kind === 'wp' && spec.parent.kind === 'wp'; $('editor-title').textContent = create ? (isSub ? 'Nouveau sous-workpackage' : 'Nouveau ' + KIND_NAMES[kind]) : 'Modifier ' + (kind === 'wp' ? 'le workpackage' : 'le ' + KIND_NAMES[kind]); var body = $('editor-body'); clear(body); var fields = {}; var input = function (id, type, value, extra) { var i = el('input', Object.assign({ type: type, id: id }, extra || {})); i.value = value || ''; fields[id] = i; return i; }; var select = function (id, options, value) { var s = el('select', { id: id }, options.map(function (o) { return el('option', { value: o.value, text: o.label }); })); s.value = value || ''; fields[id] = s; return s; }; var row = function (label, control) { return el('label', { class: 'lbl' }, [label, control]); }; body.appendChild(row('Nom', input('ed-name', 'text', create ? '' : n.name, { maxlength: '120', autocomplete: 'off' }))); if (kind === 'wp') { body.appendChild(el('div', { class: 'ed-row2' }, [ row('Code WBS', input('ed-code', 'text', create ? '' : n.code, { maxlength: '20', placeholder: create ? 'automatique' : '' })), row('Code d’imputation', input('ed-charge', 'text', create ? '' : n.charge_code, { maxlength: '40' }))])); } else if (kind === 'Livrable') { body.appendChild(el('div', { class: 'ed-row2' }, [ row('Début', input('ed-start', 'date', create ? '' : n.raw_start)), row('Fin', input('ed-finish', 'date', create ? '' : n.raw_finish))])); } else { body.appendChild(el('div', { class: 'ed-row2' }, [ row('Date', input('ed-date', 'date', create ? '' : (n.raw_finish || n.raw_start))), row('Catégorie', select('ed-cat', [{ value: '', label: '—' }, { value: 'Revue', label: 'Revue' }, { value: 'Client', label: 'Client' }, { value: 'Interne', label: 'Interne' }], create ? '' : n.milestone_category))])); } body.appendChild(row('Responsable', select('ed-owner', [{ value: '', label: 'Aucun' }].concat(people.map(function (p) { return { value: p.resource_id, label: p.name + (p.job_function ? ' — ' + p.job_function : '') }; })), create ? '' : n.owner_id))); if (!create) { var choices = parentChoices(kind, n); if (choices.length > 1) body.appendChild(row('Rattaché à', select('ed-parent', choices, kind === 'wp' ? n.parent_ref : n.wp_ref))); } $('ed-error').hidden = true; $('ed-save').disabled = false; state.editor = { spec: spec, fields: fields }; openDialog($('editor')); fields['ed-name'].focus(); }); } function submitEditor() { var ed = state.editor; if (!ed) return; var f = ed.fields, spec = ed.spec, n = spec.node || {}, kind = spec.kind, create = spec.mode === 'create'; var v = function (id) { return f[id] ? f[id].value : undefined; }; var values = { name: v('ed-name'), owner_resource_id: v('ed-owner') }; if (kind === 'wp') { values.charge_code = v('ed-charge'); if (create || v('ed-code') !== n.code) values.wbs_code = v('ed-code'); // le code n'est envoyé que s'il a changé : un déplacement renumérote sinon } else if (kind === 'Livrable') { values.planned_start = v('ed-start'); values.planned_finish = v('ed-finish'); } else { values.planned_finish = v('ed-date'); values.milestone_category = v('ed-cat'); } var req; if (create) { req = call('wbs.create', { projectId: state.scopeId, kind: kind, parentId: spec.parent && spec.parent.kind === 'wp' ? spec.parent.ref : '', values: values }, true); } else { if (f['ed-parent'] && f['ed-parent'].value !== (kind === 'wp' ? n.parent_ref : n.wp_ref)) values[kind === 'wp' ? 'parent_wp_id' : 'wp_id'] = f['ed-parent'].value; req = call('wbs.update', { kind: kind === 'wp' ? 'wp' : 'item', id: n.ref, version: n.version, patch: values }, true); } $('ed-save').disabled = true; req.then(function (r) { closeDialog($('editor')); state.pendingSelect = (r.kind === 'wp' ? 'wp:' : 'item:') + r.id; toast(create ? 'Créé : ' + r.record.name + '.' : 'Enregistré : ' + r.record.name + '.'); return load(true); }).catch(function (err) { $('ed-save').disabled = false; var m = $('ed-error'); m.textContent = err && err.code === 'CONFLICT' ? 'Cet élément a été modifié entre-temps par quelqu’un d’autre : fermez, actualisez puis recommencez.' : ppmErrorText(err); m.hidden = false; }); } function openDelete(n) { state.confirmAction = null; $('confirm-ok').textContent = 'Supprimer'; $('confirm-ok').className = 'btn danger'; var sub = n.kind === 'wp' ? below(n.id) : []; var wps = sub.filter(function (x) { return x.kind === 'wp'; }).length, items = sub.filter(function (x) { return x.kind === 'item'; }).length; var text; if (n.kind === 'item') text = '« ' + n.name + ' » sera supprimé' + (n.dep_count ? ', ainsi que ' + plural(n.dep_count, 'lien de dépendance', 'liens de dépendance') : '') + '.'; else if (wps + items) { var parts = []; if (wps) parts.push(plural(wps, 'sous-workpackage', 'sous-workpackages')); if (items) parts.push(plural(items, 'livrable ou jalon', 'livrables et jalons')); text = 'Le workpackage « ' + n.name + ' » contient ' + parts.join(' et ') + '. Tout sera supprimé, avec les liens de dépendance de ces éléments.'; } else text = 'Le workpackage « ' + n.name + ' » sera supprimé.'; $('confirm-title').textContent = 'Supprimer ' + (n.kind === 'wp' ? 'le workpackage' : 'le ' + KIND_NAMES[n.item_type]) + ' ?'; $('confirm-text').textContent = text + ' La suppression est tracée dans le journal.'; $('confirm-error').hidden = true; $('confirm-ok').disabled = false; state.deleting = { node: n, cascade: wps + items > 0 }; openDialog($('confirm')); } function confirmDelete() { if (state.confirmAction) return state.confirmAction(); var d = state.deleting; if (!d) return; $('confirm-ok').disabled = true; call('wbs.delete', { kind: d.node.kind === 'wp' ? 'wp' : 'item', id: d.node.ref, version: d.node.version, cascade: d.cascade }, true).then(function (r) { var x = r.deleted, parts = []; if (x.wps) parts.push(plural(x.wps, 'workpackage', 'workpackages')); if (x.items) parts.push(plural(x.items, 'livrable ou jalon', 'livrables et jalons')); if (x.dependencies) parts.push(plural(x.dependencies, 'dépendance', 'dépendances')); closeDialog($('confirm')); toast('Supprimé : ' + parts.join(', ') + '.'); return load(true); }).catch(function (err) { $('confirm-ok').disabled = false; var m = $('confirm-error'); m.textContent = err && err.code === 'CONFLICT' ? 'Modifié entre-temps par quelqu’un d’autre : fermez, actualisez puis recommencez.' : ppmErrorText(err); m.hidden = false; }); } $('editor-form').addEventListener('submit', function (e) { e.preventDefault(); submitEditor(); }); $('ed-cancel').addEventListener('click', function () { closeDialog($('editor')); }); $('confirm-cancel').addEventListener('click', function () { closeDialog($('confirm')); }); $('confirm-ok').addEventListener('click', confirmDelete); if (state.tab === 'wbs') $('wbs-add').addEventListener('click', function () { var root = state.data && state.data.nodes.filter(function (x) { return x.kind === 'project'; })[0]; if (root) openEditor({ mode: 'create', kind: 'wp', parent: root }); });
