// ------------------------------------------------------------ chargement function setLoading() { var c = $('canvas'); clear(c); c.appendChild(el('p', { class: 'loading', text: 'Chargement…' })); $('legend').textContent = ''; } function showError(err) { var c = $('canvas'); clear(c); c.appendChild(el('div', { class: 'empty' }, [ el('strong', { text: 'La structure n’a pas pu être chargée.' }), el('span', { text: (err && err.message) || 'Erreur inconnue.' }) ])); toast((err && err.message) || 'Erreur inconnue.', true); } function load(keep) { var ticket = ++state.seq; var scroll = keep ? { x: $('canvas').scrollLeft, y: $('canvas').scrollTop } : null; state.data = null; closeDetails(); remember(); syncControls(); // un programme : tous ses projets et leurs workpackages côte à côte (les livrables se lisent projet par projet) ; un projet : son WBS complet var req = state.scopeType === 'program' ? call('wbs.tree', { programId: state.scopeId || 'none' }) : call('wbs.tree', { projectId: state.scopeId }); setLoading(); return req.then(function (d) { if (ticket !== state.seq) return; state.data = d; setSummary(); draw(); if (scroll) { $('canvas').scrollLeft = scroll.x; $('canvas').scrollTop = scroll.y; } else initialView(); selectPending(); }).catch(function (e) { if (ticket === state.seq) showError(e); }); } function setSummary() { var d = state.data, sm = $('summary'), t = $('ztitle'), ctx = $('context'); clear(ctx); ['wbs-add', 'wbs-add-deliverable', 'wbs-add-milestone'].forEach(function (id) { $(id).hidden = !(d && d.canCreate); }); $('import-btn').hidden = !(d && d.canImport); if (!d) { sm.textContent = ''; return; } if (d.scope === 'program') { t.textContent = 'Découpage (WBS) — ' + (d.program.code ? d.program.code + ' ' : '') + d.program.name; sm.textContent = plural(d.stats.projects, 'projet', 'projets') + ', ' + plural(d.stats.wps, 'workpackage', 'workpackages') + ' ; ' + plural(d.stats.items, 'livrable ou jalon', 'livrables et jalons') + ' à lire projet par projet.'; ctx.textContent = 'Tous les projets du programme côte à côte. Cliquez sur une carte pour son détail, puis sur « Ouvrir le WBS du projet » pour modifier.'; } else { t.textContent = 'Découpage (WBS) — ' + d.project.code + ' ' + d.project.name; sm.textContent = plural(d.stats.wps, 'workpackage', 'workpackages') + ', ' + plural(d.stats.items, 'livrable ou jalon', 'livrables et jalons') + '.'; ctx.textContent = d.project.program_name ? 'Programme ' + d.project.program_name + '. ' : ''; ctx.appendChild(document.createTextNode('Cliquez sur une carte pour son détail ; le bouton sur une carte replie sa branche.')); } } // ------------------------------------------------------------ sélecteurs function fillPicker(cat) { var sel = $('picker'); clear(sel); var byProgram = {}; cat.projects.forEach(function (p) { (byProgram[p.program_id || ''] = byProgram[p.program_id || ''] || []).push(p); }); var sortCode = function (a, b) { return String(a.code).localeCompare(String(b.code), 'fr', { numeric: true }); }; var group = function (label, programId, programName) { var og = el('optgroup', { label: label }); var po = el('option', { value: 'program:' + (programId || 'none'), text: programName }); po.setAttribute('data-kind', 'program'); og.appendChild(po); (byProgram[programId || ''] || []).sort(sortCode).forEach(function (p) { og.appendChild(el('option', { value: 'project:' + p.id, text: p.code + ' ' + p.name + (p.status === 'Clos' ? ' (clos)' : '') })); }); sel.appendChild(og); }; cat.programs.slice().sort(sortCode).forEach(function (g) { group(g.code + ' ' + g.name, g.id, 'Programme ' + g.code + ' (tous les projets)'); }); if ((byProgram[''] || []).length) group('Hors programme', '', 'Projets hors programme'); } function syncControls() { var sel = $('picker'); sel.value = (state.scopeType === 'program' ? 'program:' + (state.scopeId || 'none') : 'project:' + state.scopeId); renderOptions(); }
